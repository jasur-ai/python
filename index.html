asosiy klass 
class Talaba:
    # Klass atributi (hammada bir xil)
    universitet = "TSUE"

    def __init__(self, ism, yosh):
        # Object atributlari
        self.ism = ism
        self.yosh = yosh
        self.balllar = []

    def __str__(self):
        # print() qilganda chiqadi
        return f"Talaba: {self.ism}"

    def ball_qo'sh(self, ball):
        self.balllar.append(ball)

    def o'rtacha(self):
        return sum(self.balllar) / len(self.balllar)

# Object yaratish
jasur = Talaba("Jasur", 20)
jasur.ball_qo'sh(85)
jasur.ball_qo'sh(90)
print(jasur.o'rtacha())  # 87.5
Meros (Inheritance) 
class Shaxs:          # Ota klass
    def __init__(self, ism):
        self.ism = ism

    def salomlash(self):
        return f"Salom, men {self.ism}"


class O'qituvchi(Shaxs):  # Bola klass
    def __init__(self, ism, fan):
        super().__init__(ism)  # Otani chaqirish
        self.fan = fan

    def salomlash(self):       # Override
        return f"Men {self.fan} o'qitaman"

ustoz = O'qituvchi("Akbar", "Python")
isinstance(ustoz, Shaxs)  # True!
@property, @staticmethod, @classmethod 
class Doira:
    def __init__(self, radius):
        self._radius = radius

    @property
    def radius(self):        # getter
        return self._radius

    @radius.setter
    def radius(self, r):      # setter
        if r < 0: raise ValueError()
        self._radius = r

    @staticmethod
    def pi():                 # self siz
        return 3.14159

    @classmethod
    def birlik(cls):           # klassdan
        return cls(1)
Maxfiylik + Dunder metodlar 
class Hisob:
    def __init__(self, balans):
        self.__balans = balans   # MAXFIY

    def __repr__(self):          # debug uchun
        return f"Hisob({self.__balans})"

    def __len__(self):           # len() uchun
        return self.__balans

    def __add__(self, other):    # + operator
        return Hisob(self.__balans + other.__balans)

# __dunder__ = "magic method"
# __init__ __str__ __repr__ __len__
# __add__ __eq__ __lt__ __iter__

🚨 Xatoliklarni Boshqarish 07 / EXCEPTIONS
try / except / finally 
try:
    son = int(input("Son: "))
    natija = 100 / son

except ValueError:
    print("Son kiriting!")

except ZeroDivisionError:
    print("Nolga bo'lish mumkin emas!")

except Exception as e:
    print(f"Xato: {e}")

else:
    print(f"Natija: {natija}")  # xato bo'lmasa

finally:
    print("Har doim ishlaydi")  # har doim

🌐 Virtual Muhit (venv) 08 / VIRTUAL ENV
Virtual muhit nima? — Har bir loyiha uchun alohida "quti". Bir loyihada django==4.0, boshqasida django==3.0 bo'lsa — bir-biriga xalaqit bermaydi.
venv — standart Python usuli 
# Yaratish
python3 -m venv myenv

# Faollashtirish
source myenv/bin/activate      # Linux/Mac
myenv\Scripts\activate         # Windows

# Faollashtirish belgisi:
(myenv) user@host:~$

# Kutubxona o'rnatish
pip install fastapi
pip install -r requirements.txt

# Ro'yxat saqlash
pip freeze > requirements.txt

# O'chirish
deactivate
conda — Anaconda usuli (sizda bor) 
# Muhit yaratish
conda create -n myproject python=3.11

# Faollashtirish
conda activate myproject

# O'rnatish
conda install numpy pandas
pip install fastapi     # pip ham ishlaydi

# Barcha muhitlar ro'yxati
conda env list

# Muhit eksport
conda env export > environment.yml

# Yuklab olish
conda env create -f environment.yml

# O'chirish
conda deactivate
conda remove -n myproject --all
